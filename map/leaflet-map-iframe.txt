<script>
  // ─────────────────────────────
  // CONFIG
  // ─────────────────────────────

  const veganColors = {
    'plant-based': '#F44336',
    'vegetarian': '#9C27B0',
    'vegan': '#4CAF50',
    'default': '#9E9E9E'
  };

  const mediaIcons = {
    'articles': 'https://static.wixstatic.com/shapes/f13a34_86875c6e86dc41b48dcd90f5469776e6.svg',
    'chapters': 'https://static.wixstatic.com/shapes/f13a34_6ccc8fa9603a4b7aa4acd8480eed8448.svg',
    'documentaries': 'https://static.wixstatic.com/shapes/f13a34_c4564bcda82f4ac4b9adbecc765c9270.svg',
    'movies': 'https://static.wixstatic.com/shapes/f13a34_ad11bce2d7cd4943bbecf09644d930d5.svg'
  };

  // ─────────────────────────────
  // MAP INIT
  // ─────────────────────────────

  const map = L.map('map', {
    zoomControl: false,
    minZoom: 2,
    maxZoom: 18,
    worldCopyJump: false,
    maxBoundsViscosity: 1.0,
    maxBounds: [[-85, -180], [85, 180]]
  }).setView([25.276987, 51.520008], 4);

  map.on('zoomend', () => {
    parent.postMessage({ type: 'mapZoomChanged', zoom: map.getZoom() }, '*');
  });

  L.control.zoom({ position: 'bottomright' }).addTo(map);

  L.tileLayer(
    'https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png',
    {
      attribution: '&copy; OpenStreetMap &copy; CartoDB',
      subdomains: 'abcd',
      noWrap: false,
      maxZoom: 18
    }
  ).addTo(map);

  const markersGroup = L.layerGroup().addTo(map);

  // ─────────────────────────────
  // ICON
  // ─────────────────────────────

  function createCustomIcon(article) {
    const veganity = article.category?.toLowerCase() || 'default';
    const mediaType = article.mediaType?.toLowerCase() || 'articles';
    const quality = article.qualityIndex || 0;

    const arcPath = describeArc(20, 20, 18, 0, -(quality * 180));

    const svg = `
      <svg class="marker-svg" viewBox="0 0 60 90" xmlns="http://www.w3.org/2000/svg">

        <path d="M20,2 
                 C30,2 38,10 38,20 
                 C38,32 20,58 20,58 
                 C20,58 2,32 2,20 
                 C2,10 10,2 20,2 Z"
              fill="${veganColors[veganity]}"
              stroke="white"
              stroke-width="2"/>

        <path d="${arcPath}"
              fill="none"
              stroke="black"
              stroke-width="6"
              stroke-linecap="round"/>

        <path class="quality-arc" d="${arcPath}"/>

        <image href="${mediaIcons[mediaType] || mediaIcons.articles}"
               x="7" y="10" width="26" height="26" />

      </svg>
    `;

    return L.divIcon({
      html: svg,
      className: '',
      iconSize: [60, 90],
      iconAnchor: [30, 58],
      popupAnchor: [0, -58]
    });
  }

  // ─────────────────────────────
  // GEOMETRY
  // ─────────────────────────────

  function describeArc(cx, cy, r, startAngle, endAngle) {
    const start = polarToCartesian(cx, cy, r, startAngle);
    const end = polarToCartesian(cx, cy, r, endAngle);

    const largeArc = Math.abs(endAngle - startAngle) > 180 ? 1 : 0;
    const sweep = endAngle < startAngle ? 0 : 1;

    return `M ${start.x} ${start.y} A ${r} ${r} 0 ${largeArc} ${sweep} ${end.x} ${end.y}`;
  }

  function polarToCartesian(cx, cy, r, angle) {
    const rad = angle * Math.PI / 180;
    return {
      x: cx + r * Math.cos(rad),
      y: cy + r * Math.sin(rad)
    };
  }

  // ─────────────────────────────
  // MARKERS
  // ─────────────────────────────

  function renderMarkers(articles) {
    markersGroup.clearLayers();

    const bounds = [];
    const positions = {};

    articles.forEach(article => {
      if (!article.latitude || !article.longitude) return;

      let lat = Number(article.latitude);
      let lng = Number(article.longitude);
      if (isNaN(lat) || isNaN(lng)) return;

      const key = `${lat.toFixed(5)}_${lng.toFixed(5)}`;
      positions[key] = positions[key] || [];

      const index = positions[key].length;
      positions[key].push(article._id);

      if (index > 0) {
        const radius = 0.03;
        const angle = index * 45;

        lat += radius * Math.cos(angle);
        lng += radius * Math.sin(angle);
      }

      const marker = L.marker([lat, lng], {
        icon: createCustomIcon(article)
      });

      marker.on('mouseover', () => {
        window.parent.postMessage({
          articleId: article.id,
          action: 'hover'
        }, '*');
      });

      marker.on('click', () => {
        window.parent.postMessage({
          articleId: article.id,
          action: 'click'
        }, '*');
      });

      marker.addTo(markersGroup);
      bounds.push([lat, lng]);
    });

    if (bounds.length) {
      map.fitBounds(bounds, { padding: [50, 50] });
    }
  }

  // ─────────────────────────────
  // MESSAGE HANDLER
  // ─────────────────────────────

  window.addEventListener("message", (event) => {
    const data = event.data;
    if (!data) return;

    if (Array.isArray(data)) {
      renderMarkers(data);
      return;
    }

    if (data.action === "centerMarker") {
      const lat = Number(data.latitude);
      const lng = Number(data.longitude);
      if (!isNaN(lat) && !isNaN(lng)) {
        map.setView([lat, lng], Math.max(map.getZoom(), 6), {
          animate: true
        });
      }
      return;
    }

    if (data.type === "hoverMarker") {
      highlightMarker(data.articleId);
    }

    if (data.type === "unhoverMarker") {
      unhighlightMarker(data.articleId);
    }
  });

</script>
